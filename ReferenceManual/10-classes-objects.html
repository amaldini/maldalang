10. Classes & Objects

10.1 Class Declaration

class ClassName {
    public var field1;
    private var field2;
    
    function ClassName(param1, param2) {
        this.field1 = param1;
        this.field2 = param2;
    }
    
    public function method1() {
        field1 = "value";  // Implicit member access
    }
    
    private function method2() {
        // method body
    }
}

10.2 Access Modifiers

  • public: Accessible from anywhere (default for methods)
  • private: Accessible only within the class
  • Fields default to private if no modifier is specified

10.3 Constructors

Constructors have the same name as the class and are called automatically when objects are created:

class Point {
    public var x;
    public var y;
    
    function Point(x, y) {
        this.x = x;
        this.y = y;
    }
}

10.4 Methods

Methods are functions defined within classes. They can access class fields using this or implicitly:

class Calculator {
    public var result;
    
    function Calculator() {
        result = 0;  // Implicit member access
    }
    
    public function add(value) {
        result = result + value;
        return result;
    }
}

10.5 Object Instantiation

var obj = new ClassName(arg1, arg2);
var person = new Person("Alice", 25);

10.6 Member Access

var person = new Person("Bob", 30);
person.name = "Charlie";
var age = person.age;
person.setAge(31);

10.7 The this Keyword

this refers to the current object instance. It's optional when accessing member variables (implicit member access is supported):

class Counter {
    private var count;
    
    function Counter() {
        count = 0;  // Implicit member access
    }
    
    public function setCount(count) {
        this.count = count;  // 'this.' required due to name conflict
    }
}

10.8 Inheritance

Classes can inherit from a parent class using extends:

class Animal {
    public var name;
    
    function Animal(name) {
        this.name = name;
    }
    
    public function speak() {
        print(name + " makes a sound");
    }
}

class Dog extends Animal {
    function Dog(name) {
        super(name);  // Call parent constructor
    }
    
    public function speak() {
        super.speak();  // Call parent method
        print($"{name} also barks");
    }
}

10.9 Static Members

Static members belong to the class, not instances:

class MathUtils {
    static var PI = 3.14159;
    
    static function add(a, b) {
        return a + b;
    }
}

var pi = MathUtils.PI;
var sum = MathUtils.add(5, 3);

10.10 Operator Overloading

Classes can overload selected operators by defining methods with reserved names. For binary operators, MALDA first tries the left operand overload (__add__, __sub__, etc.). If that overload is not found, MALDA then tries the right operand reversed overload (__radd__, __rsub__, etc.) with the left operand passed as argument. For unary operators, the single operand is the receiver.

OperatorLeft-hand methodRight-hand fallback method
+__add__(other)__radd__(other)
- (binary)__sub__(other)__rsub__(other)
*__mul__(other)__rmul__(other)
/__div__(other)__rdiv__(other)
%__mod__(other)__rmod__(other)
==__eq__(other)__req__(other)
!=__neq__(other)__rneq__(other)
<__lt__(other)__rlt__(other)
<=__le__(other)__rle__(other)
>__gt__(other)__rgt__(other)
>=__ge__(other)__rge__(other)
- (unary)__neg__()n/a

If no matching left-hand or right-hand overload method exists, MALDA uses the built-in operator behavior.

class Value {
    public var data;

    function Value(data) {
        this.data = data;
    }

    public function __add__(other) {
        return data + other.data;
    }
}

var a = new Value(2);
var b = new Value(3);
print(a + b); // 5
class RightValue {
    public var data;

    function RightValue(data) {
        this.data = data;
    }

    public function __radd__(other) {
        return other + data;
    }
}

var rhs = new RightValue(7);
print(5 + rhs); // 12
