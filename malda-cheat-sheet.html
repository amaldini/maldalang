1) Fundamentals

Variables, interpolation, null-safe access. Prefer io.print outside actors.

var name = "MALDA";
var version = 1;
const pi = 3.14159;
var isAwesome = true;
var nothing = null;

io.print($"Hello from {name} v{version}");
io.print("2 + 2 = " + string(2 + 2));

var user = {"id": 1, "name": "Ada", "active": true};
io.print(user.name);
io.print(user?.email ?? "no email");

2) Control Flow + Pattern Matching

if/else, C-style for, foreach, match/case. for (var x in arr) is the same as foreach.

var x = 7;
if (x > 10) {
  io.print("big");
} else {
  io.print("small");
}

for (var i = 0; i < 3; i = i + 1) {
  io.print(i);
}

var arr = [10, 20, 30];
foreach (var item in arr) {
  io.print(item);
}

match x {
  case 0: io.print("zero");
  case 7: io.print("seven");
  default: io.print("other");
}

2b) Advanced Match Patterns

Wildcard, bindings, nested object/array patterns, and sum-type variants.

var data = [{ type: "A", value: 1 }, { type: "B", value: 2 }];
var m1 = match data {
  case [{ type: "A", value: v }, ...rest]: "first=" + v;
  default: "none";
};
io.print(m1);

var profile = { name: "Alice", age: 30, city: "Rome" };
var m2 = match profile {
  case { name, age }: name + " is " + age;
  case { name }: name;
  case _: "unknown";
};
io.print(m2);

type Result = Ok(value) | Err(message);
var r = Ok(42);
var m3 = match r {
  case Ok(v): "ok: " + v;
  case Err(msg): "error: " + msg;
};
io.print(m3);

3) Functions + Lambdas

Use function — fn and def are syntax errors. Typed params are optional on regular functions, never on prompts.

function sum(a, b) {
  return a + b;
}

function applyTwice(value, fn) {
  return fn(fn(value));
}

var inc = (n) => n + 1;
var result = applyTwice(3, inc);
io.print(result); // 5

function greet(name: string) {
  io.print("Hello " + name);
}

3b) Modules

import merges exports; selective import takes only named ones. include splices source into the host.

import { add, VERSION } from "helpers/math_lib.malda";
io.print(add(2, 3));

// File module: isolated env; only export bindings merge.
import "helpers/math_lib.malda";

// include splices source (shared globals). Prefer import
// when the library is self-contained.
include "helpers/host_contract.malda";

4) Classes + Objects

Constructors, methods, inheritance, this/super. A parameter list after the class name is a primary constructor — do not mix with extends.

class Animal {
  var name;
  function Animal(name) {
    this.name = name;
  }
  function speak() {
    return "...";
  }
}

class Dog extends Animal {
  function Dog(name) {
    super(name);
  }
  function speak() {
    return $"Woof! I'm {this.name}";
  }
}

var d = new Dog("Rex");
io.print(d.speak());

class Point(x, y) {
  function total() {
    return this.x + this.y;
  }
}

5) Arrays + Maps + JSON

.length is a property, not a call. Arrays use member methods — there is no arr namespace.

var nums = [1, 2, 3];
nums.append(4);
io.print(nums.length); // 4

var profile = {"name": "Lina", "age": 28};
profile.city = "Milan";
io.print(profile.city);

var payload = parseJSON("{\"ok\": true, \"count\": 2}");
io.print(payload.ok);
io.print(toJSON(profile));

var [a, b, ...rest] = nums;
io.print(a + ", " + b);

var {name: personName, city: personCity} = profile;
io.print(personName + " from " + personCity);

var {name, city} = profile;
io.print(name + " - " + city);

6) Built-ins + File/OS

Prefer namespaced calls. Flat aliases still run but the language server warns. io.getEnv returns null when unset.

var s = string(123);
var n = int("42");
var f = float("2.5");

io.print(str.length("hello"));
io.print(str.substring("abcdef", 1, 3)); // "bcd"

var key = io.getEnvOr("OPENAI_API_KEY");
var here = getProgramDirectory();
io.print(here);

io.writeFile("out.txt", "Hello");
var txt = io.readFile("out.txt");
io.print(txt);

var cmd = runCommand("dotnet --version");
io.print(cmd);

7) Error Handling

Throw/catch/finally patterns and guard checks.

function divide(a, b) {
  if (b == 0) {
    throw "Division by zero";
  }
  return a / b;
}

try {
  var z = divide(10, 0);
  io.print(z);
} catch (e) {
  io.print("Error: " + string(e));
} finally {
  io.print("cleanup done");
}

8) Actors + Concurrency

Handlers do not see io / math / str — use flat print inside on. Outside the actor, prefer io.print.

actor Counter {
  var count = 0;

  on increment() {
    count = count + 1;
    print($"count={count}");
  }

  on get() {
    print($"current={count}");
  }
}

var c = spawn Counter();
sleep(100);
send c.increment();
send c.increment();
send c.get();
sleep(500);

9) Prompt Blocks + Schema

Name-only prompt params. -> Type on await validates JSON against a declared schema or sum type — it is not compile-time typing.

schema Plan {
  steps: string[];
}

prompt greet(name) {
  user: "Hello {name}, how can I help?"
}

prompt planTask(task, docs) -> Plan {
  system: "You are a planner.",
  user: """
  Task: {task}
  Docs:
  {docs}
  Return JSON only.
  """
}

var p = planTask("Add auth", "README + API docs");
io.print(p.user); // prompt(...) returns PromptInstance

var plan = await planTask("Add auth", "README + API docs");
io.print(plan.steps[0]);

11) Web UI + HttpServer

@PAGE returns HTML strings. Component trees use ui.* on page 2 — do not mix the two models on one surface.

var server = new HttpServer(8080);

@PAGE("/")
function home() {
  return "<h1>Welcome to MALDA</h1>";
}

@PAGE("/user/{id}")
function userPage(id) {
  return "<h2>User " + id + "</h2>";
}

@POST("/submit")
function submit(body) {
  return {"status": 200, "message": "ok", "received": body};
}

server.start();

12) REST API Server + HTTP Client

API-only: new RestServer(8080). Fullstack: port-less RestServer + http.mount(api) on the HttpServer that owns the port.

@GET("/api/users/{id}")
function getUserById(id) {
  return {"id": id, "name": "Alice"};
}

@POST("/api/users")
function createUser(body) {
  return {"status": 201, "data": body};
}

var api = new RestServer();
var http = new HttpServer(8080);
http.mount(api);
http.start();

var client = new RestClient("https://api.example.com");
client.setAuth("Bearer", "token");
var r = client.get("/users", null, {"limit": "10"});
if (r.ok) io.print(toJSON(r.body));

13) Database Clients

SQL Server, PostgreSQL, SQLite with query + fluent CRUD support.

var db = new SqliteClient();
db.connect("Data Source=app.db;");

var rows = db.query(
  "SELECT * FROM users WHERE age > @age",
  {age: 25}
);

var result = db.select("name", "email")
  .from("users")
  .where("age", ">", 25)
  .query();

db.insert("users", {"name": "John", "email": "john@example.com"});
db.disconnect();

14) MCP Server + Client

Expose MALDA functions as tools; call another server with MCPClient.

schema AddArgs {
  a: int;
  b: int;
}

@MCPTool("add", "Adds two numbers", "AddArgs")
function add(a, b) {
  return int(a) + int(b);
}

var mcpServer = new MCPServer();
var listed = mcpServer.getTools();
io.print(listed[0].inputSchema.properties.a.type);
var local = mcpServer.callTool("add", {"a": 5, "b": 3});
io.print(local.data);

var mcp = new MCPClient("example-server");
var ok = mcp.connect("malda", ["server.malda"]);
if (ok) {
  var sum = mcp.callTool("add", {"a": 5, "b": 3});
  io.print(sum);
  mcp.disconnect();
}

15) ACP (Agent Communication Protocol)

Agent-to-agent interoperability across systems.

var agents = acp.discoverAgents();
var reply = acp.sendMessage("translation-agent", "Translate hello to Spanish", 5000);
io.print(reply);

var llm = new OpenRouterClient();

var server = new ACPServer(8080);

  "name": "Helper",

16) Graphs

First-class graph directed / graph undirected literals — not a JSON object.

var g = graph directed {
  nodes: ["A", "B", "C"],
  edges: [
    { from: "A", to: "B", weight: 3 },
    { from: "B", to: "C", weight: 2 }
  ]
};

io.print(g.nodeCount());
io.print(g.getNeighbors("A"));
io.print(g.getWeight("A", "B"));
g.addNode("D");
g.addEdge("C", "D", 1);

16b) GraphMemory

Semantic memory for agents. Call initialize() before remember/query. Opt-in citations: memory.ask or query(..., { grounded: true }).

var memory = new GraphMemory();
memory.initialize();

memory.remember("MALDA uses actors for concurrency");
memory.remember("VectorDB stores embeddings");

var hits = memory.query("How does concurrency work?", 3);
io.print(hits[0]);

17) VectorDB + Semantic Search

Constructor needs dimension + precision. init(fn) then add(data) / searchSimilar(data, topN). Hits expose .data and .similarity.

function embed(text) {
  return [1.0, 0.0, 0.0];
}

var db = new VectorDB(3, "double");
db.init(embed);
db.add("MALDA is an AI-first language");
db.add("Actors handle concurrency");

var hits = db.searchSimilar("How do actors work?", 3);
foreach (var hit in hits) {
  io.print(string(hit.data) + " sim=" + string(hit.similarity));
}

Load assemblies, instantiate types, call methods/properties.

MALDA UI Framework Cheat Sheet

Focused reference for server-side UI composition in MALDA using ui.*, runtime lifecycle APIs, event pipeline, forms, routing/session helpers, and schema-driven CRUD helpers.

Components Runtime + Lifecycle Events + Forms Routing + Session Templates + CRUD Data Controls

UI 1) Component Composition

Build a tree with layout/display/input controls using ui.control(props, children?, key?).

function itemActions() {
  return ui.row({"gap": 8}, [
    ui.button({"label": "Assign", "variant": "primary", "onClick": "assignItem"}),
    ui.button({"label": "Close", "variant": "danger", "onClick": "closeItem"})
  ]);
}

var itemCard = ui.panel({"componentId": "ItemCard", "className": "card"}, [
  ui.heading({"level": 2, "value": "Item #123"}),
  ui.text({"value": "Login issue reported by user"}),
  itemActions()
]);

var root = ui.column({"gap": 12}, [
  ui.heading({"level": 1, "value": "Items Dashboard"}),
  itemCard
]);

UI 2) Runtime + Lifecycle

Mount/render, snapshot/resync, state helpers, and lifecycle event hooks.

var sessionId = "app-ui";

ui.configure("maxPayloadBytes", 250000, sessionId);
ui.configure("sessionTtlMs", 1800000, sessionId);

var envelope = ui.mountEnvelope(root, sessionId, {"includeSnapshot": true});

var wasOpen = ui.state("ItemPanel", "open", false, "session");
ui.setState("ItemPanel", "open", !wasOpen, "session");

ui.onInit("ItemPanel", sessionId);
ui.onPreRender("ItemPanel", sessionId);
ui.onUpdate("ItemPanel", sessionId);
ui.onError("ItemPanel", sessionId);

ui.invalidate("items", {"reason": "server-update"});
var ev = ui.pullEvent(sessionId);

UI 3) Events + Form Patterns

Drain with ui.pullEvent before rebuilding. Second arg of dispatchEvent is sessionId (string), not a sequence number.

var form = ui.form({"componentId": "CreateItemForm", "onSubmit": "submitItem"}, [
  ui.field({}, [
    ui.text({"value": "Title"}),
    ui.textField({"name": "title", "required": true, "onChange": "validateTitle"})
  ]),
  ui.field({}, [
    ui.text({"value": "Priority"}),
    ui.select({"name": "priority", "options": [
      {"label": "Low", "value": "low"},
      {"label": "High", "value": "high"}
    ]})
  ]),
  ui.button({"label": "Create", "variant": "primary", "onClick": "submit"})
]);

function validateTitle(event) {
  if (event.value == null || str.length(event.value) < 3) {
    ui.setState("CreateItemForm", "titleError", "Title must be at least 3 chars", "session");
  } else {
    ui.setState("CreateItemForm", "titleError", "", "session");
  }
}

function submitItem(event) {
  ui.setState("CreateItemForm", "saved", true, "session");
}

ui.dispatchEvent({"type": "submit", "componentId": "CreateItemForm", "values": {"title": "VPN issue"}}, "app-ui");
var pending = ui.pullEvent("app-ui");

UI 4) Routing + Session Helpers

Preserve user UI session across server routes and redirects.

@GET("/web/items")
function itemsPage(req) {
  var sessionId = ui.sessionId(req, "items-ui");
  var app = ui.column({}, [
    ui.heading({"level": 1, "value": "Items"}),
    ui.text({"value": "Filter and manage items"})
  ]);
  return ui.mountEnvelope(app, sessionId, {"includeSnapshot": true});
}

@POST("/web/items/create")
function createItem(body) {
  return ui.redirectWithSession("/web/items", body, "items-ui");
}

UI 5) Templates + CRUD Helpers

Compose templates and generate schema-driven CRUD controls quickly.

var base = "templates/";

var sidebar = ui.template(base + "sidebar.html", {"active": "items"}, {"cache": false});
var body = ui.partial(base + "items_body.html", {"rowsHtml": rowsHtml});
var page = ui.layout(base + "layout.html", {"sidebar": sidebar, "body": body}, {"title": "App"});

var schema = ui.crudSchema({
  "entityName": "Item",
  "entitySingularLower": "item",
  "columns": ["id", "title", "priority"],
  "filterDefs": [{"name": "priority", "type": "select"}]
});

var controlsHtml = ui.crudControls(
  schema,
  "app-session",
  {"page": "1"},
  {"owners": []},
  {"cache": false}
);
